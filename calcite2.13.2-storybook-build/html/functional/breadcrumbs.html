Breadcrumbs Functional Component

Use the <Breadcrumbs /> component to display a list of links that represent the current page's location within a hierarchy.

Examples

  • Workspace Header

Usage

The following shows how to render breadcrumbs in a secondary navigation menu with optional icons and click handler:

// renders "Home > Documents > Title of Current Document" with icons
get breadcrumbs(): IHubBreadcrumb[] {
  return [
    { title: 'Home', icon: 'home', link: '/' },
    { title: 'Documents', icon: 'folder', link: '/documents' },
    // the last one should be the current page and will be rendered as "active"
    // and any that do not have a link will have the "non-interactive" CSS class applied
    { title: 'Title of Current Document', icon: 'file-text' }
  ]
}

// NOTE: click handler is optional
handleBreadcrumbClick = (clickEvent): void => {
  const href = clickEvent.target.href;
  console.log(href);
}

// NOTE: this should be rendered w/in a calcite-menu
renderSecondaryNav () {
  return (<calcite-navigation slot="navigation-secondary">
    <calcite-menu scale="s" slot="content-start">
      <Breadcrumbs breadcrumbs={this.breadcrumbs} onClick={this.handleBreadcrumbClick} />
    </calcite-menu>
  </calcite-navigation>)
}
